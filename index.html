mport random import time import turtle # Настройка экрана window = turtle.Screen() window.title("Игра Змейка на Python") window.bgcolor("black") window.setup(width=600, height=600) window.tracer(0) # Голова змейки snake_head = turtle.Turtle() snake_head.speed(0) snake_head.shape("square") snake_head.color("green") snake_head.penup() snake_head.goto(0, 0) snake_head.direction = "stop" # Тело змейки snake_segments = [] # Еда для змейки food = turtle.Turtle() food.speed(0) food.shape("circle") food.color("red") food.penup() food.goto(0, 100) # Счет score = 0 high_score = 0 score_display = turtle.Turtle() score_display.speed(0) score_display.color("white") score_display.penup() score_display.hideturtle() score_display.goto(0, 260) score_display.write("Счет: 0 Рекорд: 0", align="center", font=("Courier", 14, "normal")) # Функции управления def go_up(): if snake_head.direction != "down": snake_head.direction = "up" def go_down(): if snake_head.direction != "up": snake_head.direction = "down" def go_left(): if snake_head.direction != "right": snake_head.direction = "left" def go_right(): if snake_head.direction != "left": snake_head.direction = "right" def move(): if snake_head.direction == "up": y = snake_head.ycor() snake_head.sety(y + 20) if snake_head.direction == "down": y = snake_head.ycor() snake_head.sety(y - 20) if snake_head.direction == "left": x = snake_head.xcor() snake_head.setx(x - 20) if snake_head.direction == "right": x = snake_head.xcor() snake_head.setx(x + 20) # Привязка клавиш window.listen() window.onkeypress(go_up, "w") window.onkeypress(go_down, "s") window.onkeypress(go_left, "a") window.onkeypress(go_right, "d") window.onkeypress(go_up, "Up") window.onkeypress(go_down, "Down") window.onkeypress(go_left, "Left") window.onkeypress(go_right, "Right") # Главный игровой цикл delay = 0.1 while True: window.update() # Проверка столкновения с краями экрана if ( snake_head.xcor() > 290 or snake_head.xcor() < -290 or snake_head.ycor() > 290 or snake_head.ycor() < -290 ): time.sleep(1) snake_head.goto(0, 0) snake_head.direction = "stop" for segment in snake_segments: segment.goto(1000, 1000) snake_segments.clear() score = 0 score_display.clear() score_display.write( f"Счет: {score} Рекорд: {high_score}", align="center", font=("Courier", 14, "normal"), ) # Поедание еды if snake_head.distance(food) < 20: x = random.randint(-270, 270) y = random.randint(-270, 270) food.goto(x, y) new_segment = turtle.Turtle() new_segment.speed(0) new_segment.shape("square") new_segment.color("orange") new_segment.penup() snake_segments.append(new_segment) score += 10 if score > high_score: high_score = score score_display.clear() score_display.write( f"Счет: {score} Рекорд: {high_score}", align="center", font=("Courier", 14, "normal"), ) # Движение сегментов тела for index in range(len(snake_segments) - 1, 0, -1): x = snake_segments[index - 1].xcor() y = snake_segments[index - 1].ycor() snake_segments[index].goto(x, y) if len(snake_segments) > 0: x = snake_head.xcor() y = snake_head.ycor() snake_segments[0].goto(x, y) move() # Проверка столкновения с собственным хвостом for segment in snake_segments: if segment.distance(snake_head) < 20: time.sleep(1) snake_head.goto(0, 0) snake_head.direction = "stop" for s in snake_segments: s.goto(1000, 1000) snake_segments.clear() score = 0 score_display.clear() score_display.write( f"Счет: {score} Рекорд: {high_score}", align="center", font=("Courier", 14, "normal"), ) time.sleep(delay)
